import 'package:cloud_firestore/cloud_firestore.dart';
import 'package:firebase_auth/firebase_auth.dart';
import 'package:flutter/material.dart';

import 'form_widgets.dart';
import 'passenger_home.dart' show RPColors;
import '../services/profile_photo_service.dart';

// Firestore field names in users/{uid}. Change these if your schema differs.
const String _kNameField = 'name';

/// Lets the user edit their name and phone number.
/// Pops with `{'name': ..., 'phone': ...}` when saved, so the caller can
/// refresh what it shows.
class EditProfileScreen extends StatefulWidget {
  const EditProfileScreen({
    super.key,
    required this.initialName,
    required this.initialPhone,
    required this.email,
    this.avatarImageUrl,
  });

  final String initialName;
  final String initialPhone;
  final String email;
  final String? avatarImageUrl;

  @override
  State<EditProfileScreen> createState() => _EditProfileScreenState();
}

class _EditProfileScreenState extends State<EditProfileScreen> {
  final _formKey = GlobalKey<FormState>();
  late final TextEditingController _nameController;
  late final TextEditingController _phoneController;
  bool _saving = false;
  bool _uploadingPhoto = false;
  String? _avatarUrl;

  @override
  void initState() {
    super.initState();
    _avatarUrl = widget.avatarImageUrl;
    _nameController = TextEditingController(text: widget.initialName);
    _phoneController = TextEditingController(text: widget.initialPhone);
  }

  @override
  void dispose() {
    _nameController.dispose();
    _phoneController.dispose();
    super.dispose();
  }

  String? _validateName(String? value) {
    final v = (value ?? '').trim();
    if (v.isEmpty) return 'Please enter your name.';
    if (v.length < 2) return 'Name is too short.';
    return null;
  }

  Future<void> _save() async {
    FocusScope.of(context).unfocus();
    if (!_formKey.currentState!.validate()) return;

    final uid = FirebaseAuth.instance.currentUser?.uid;
    if (uid == null) {
      _showMessage('You are not signed in.');
      return;
    }

    final name = _nameController.text.trim();

    setState(() => _saving = true);

    try {
      await FirebaseFirestore.instance.collection('users').doc(uid).set({
        _kNameField: name,
      }, SetOptions(merge: true));

      if (!mounted) return;
      _showMessage('Profile updated.');
      Navigator.of(context).pop();
    } catch (e) {
      debugPrint('Edit profile error: $e');
      if (!mounted) return;
      setState(() => _saving = false);
      _showMessage('Could not save your changes. Please try again.');
    }
  }

  Future<void> _changePhoto() async {
    if (_uploadingPhoto) return;
    setState(() => _uploadingPhoto = true);

    final result = await changeProfilePhoto(
      context,
      hasPhoto: (_avatarUrl ?? '').isNotEmpty,
    );

    if (!mounted) return;
    setState(() {
      _uploadingPhoto = false;
      if (result != null) _avatarUrl = result.isEmpty ? null : result;
    });
  }

  void _showMessage(String text) {
    ScaffoldMessenger.of(context).showSnackBar(SnackBar(content: Text(text)));
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      backgroundColor: RPColors.bg,
      appBar: rpAppBar('Edit Profile'),
      body: SafeArea(
        child: SingleChildScrollView(
          keyboardDismissBehavior: ScrollViewKeyboardDismissBehavior.onDrag,
          padding: const EdgeInsets.fromLTRB(20, 16, 20, 24),
          child: Form(
            key: _formKey,
            child: Column(
              crossAxisAlignment: CrossAxisAlignment.start,
              children: [
                Center(child: _buildAvatar()),
                const SizedBox(height: 24),
                Container(
                  padding: const EdgeInsets.all(16),
                  decoration: BoxDecoration(
                    color: RPColors.cardBg,
                    borderRadius: BorderRadius.circular(14),
                    border: Border.all(color: RPColors.border),
                  ),
                  child: Column(
                    crossAxisAlignment: CrossAxisAlignment.start,
                    children: [
                      const RPFieldLabel('Full Name'),
                      TextFormField(
                        controller: _nameController,
                        textCapitalization: TextCapitalization.words,
                        textInputAction: TextInputAction.done,
                        decoration: rpInputDecoration(hint: 'Your full name'),
                        validator: _validateName,
                        onFieldSubmitted: (_) => _save(),
                      ),
                      const SizedBox(height: 16),
                      const RPFieldLabel('Phone Number'),
                      TextFormField(
                        controller: _phoneController,
                        readOnly: true,
                        style: const TextStyle(color: RPColors.textGrey),
                        decoration: rpInputDecoration(readOnly: true),
                      ),
                      const SizedBox(height: 6),
                      const Text(
                        'Your number is verified. Contact support to change it.',
                        style: TextStyle(
                          fontSize: 11,
                          color: RPColors.textGrey,
                        ),
                      ),
                      const SizedBox(height: 16),
                      const RPFieldLabel('Email'),
                      TextFormField(
                        initialValue: widget.email,
                        readOnly: true,
                        style: const TextStyle(color: RPColors.textGrey),
                        decoration: rpInputDecoration(readOnly: true),
                      ),
                      const SizedBox(height: 6),
                      const Text(
                        'Email is your login and can\'t be changed here.',
                        style: TextStyle(
                          fontSize: 11,
                          color: RPColors.textGrey,
                        ),
                      ),
                    ],
                  ),
                ),
                const SizedBox(height: 24),
                RPPrimaryButton(
                  label: 'Save Changes',
                  icon: Icons.check_circle_outline,
                  loading: _saving,
                  onPressed: _save,
                ),
              ],
            ),
          ),
        ),
      ),
    );
  }

  Widget _buildAvatar() {
    final url = _avatarUrl;
    return SizedBox(
      width: 96,
      height: 96,
      child: Stack(
        children: [
          Container(
            width: 96,
            height: 96,
            padding: const EdgeInsets.all(5),
            decoration: BoxDecoration(
              shape: BoxShape.circle,
              border: Border.all(color: RPColors.teal, width: 3),
            ),
            child: CircleAvatar(
              backgroundColor: RPColors.teal,
              backgroundImage: url != null ? NetworkImage(url) : null,
              child: _uploadingPhoto
                  ? const CircularProgressIndicator(
                      strokeWidth: 3,
                      color: Colors.white,
                    )
                  : (url == null
                        ? const Icon(
                            Icons.person,
                            color: Colors.white,
                            size: 40,
                          )
                        : null),
            ),
          ),
          Positioned(
            right: 0,
            bottom: 0,
            child: InkWell(
              onTap: _changePhoto,
              borderRadius: BorderRadius.circular(20),
              child: Container(
                padding: const EdgeInsets.all(7),
                decoration: BoxDecoration(
                  color: Colors.white,
                  shape: BoxShape.circle,
                  border: Border.all(color: RPColors.teal, width: 1.5),
                ),
                child: const Icon(
                  Icons.camera_alt,
                  size: 16,
                  color: RPColors.teal,
                ),
              ),
            ),
          ),
        ],
      ),
    );
  }
}
